cmake_minimum_required(VERSION 3.16)
project(StoneKV VERSION 1.0.0 LANGUAGES CXX)

option(STONEKV_BUILD_TESTS "Build test suite" ON)
option(STONEKV_ENABLE_SANITIZERS "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(STONEKV_ENABLE_TSAN "Build with ThreadSanitizer (separate from ASan/UBSan)" OFF)
if(STONEKV_ENABLE_SANITIZERS AND STONEKV_ENABLE_TSAN)
  message(FATAL_ERROR "ThreadSanitizer and AddressSanitizer must use separate builds")
endif()

add_library(stonekv STATIC src/db.cpp)
target_include_directories(stonekv PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(stonekv PUBLIC cxx_std_17)
target_compile_options(stonekv PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(STONEKV_ENABLE_SANITIZERS)
  target_compile_options(stonekv PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(stonekv PUBLIC -fsanitize=address,undefined)
endif()
if(STONEKV_ENABLE_TSAN)
  target_compile_options(stonekv PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(stonekv PUBLIC -fsanitize=thread)
endif()

add_executable(stonekv_cli tools/cli.cpp)
target_link_libraries(stonekv_cli PRIVATE stonekv)
add_executable(stonekv_bench tools/bench.cpp)
target_link_libraries(stonekv_bench PRIVATE stonekv)
find_package(Threads REQUIRED)
target_link_libraries(stonekv PUBLIC Threads::Threads)
add_executable(stonekv_benchmark_workloads tools/benchmark_workloads.cpp)
target_compile_definitions(stonekv_benchmark_workloads PRIVATE STONEKV_EXTENDED_STATS)
target_link_libraries(stonekv_benchmark_workloads PRIVATE stonekv Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_executable(stonekv_io_profile tools/io_profile.cpp)
  target_compile_definitions(stonekv_io_profile PRIVATE STONEKV_EXTENDED_STATS)
  target_link_libraries(stonekv_io_profile PRIVATE stonekv Threads::Threads)
  target_link_options(stonekv_io_profile PRIVATE -Wl,--wrap=write -Wl,--wrap=pwrite -Wl,--wrap=pread)
endif()
add_executable(stonekv_sst_index_probe tools/sst_index_probe.cpp)
target_compile_definitions(stonekv_sst_index_probe PRIVATE STONEKV_HAS_INDEX_STATS)
target_link_libraries(stonekv_sst_index_probe PRIVATE stonekv Threads::Threads)
